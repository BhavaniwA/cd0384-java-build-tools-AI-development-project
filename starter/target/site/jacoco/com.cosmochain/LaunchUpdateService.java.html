package com.cosmochain;

import java.util.List;
import java.util.Objects;

import com.cosmochain.support.Launch;
import com.cosmochain.support.LaunchRepository;
import com.cosmochain.support.Notifier;
import com.cosmochain.support.RateLimitException;
import com.cosmochain.support.UpcomingLaunchClient;

public class LaunchUpdateService {
    private final List<UpcomingLaunchClient> providers;
    private final LaunchRepository repo;
    private final Notifier notifier;

    /**
     * Creates a launch update service with provider clients and state dependencies.
     *
     * @param providers provider clients processed by the workflow.
     * @param repo repository storing last-seen launch state.
     * @param notifier notifier used when updates are detected.
     */
    public LaunchUpdateService(List<UpcomingLaunchClient> providers, LaunchRepository repo, Notifier notifier) {
        this.providers = List.copyOf(providers);
        this.repo = repo;
        this.notifier = notifier;
    }

    /**
     * Checks all providers for updates and isolates per-provider failures.
     */
    public void checkForUpdatesAcrossProviders() {
        for (int i = 0; i < providers.size(); i++) {
            checkForUpdate(providers.get(i));

            if (i < providers.size() - 1) {
                System.out.println();
            }
        }
    }

    /**
     * Checks one provider for a change and updates state/notifications when needed.
     *
     * @param client provider client used for retrieving the next launch.
     */
    void checkForUpdate(UpcomingLaunchClient client) {
        String provider = resolveProviderName(client);
        if (provider == null) {
            return;
        }

        try {
            Launch next = client.fetchNextLaunch();
            Launch last = repo.getLastSeen(provider);

            if (last == null) {
                repo.setLastSeen(provider, next);
                notifier.notify(provider, "NEW", next);
            } else if (!Objects.equals(last.id, next.id) || !Objects.equals(last.dateUtc, next.dateUtc)) {
                String changeType = !Objects.equals(last.id, next.id) ? "NEW" : "DATE_CHANGED";
                repo.setLastSeen(provider, next);
                notifier.notify(provider, changeType, next);
            } else {
                System.out.println("No changes detected.");
            }
        } catch (RateLimitException e) {
            System.out.printf(
                "Skipping %s: API throttled. Retry in %d seconds.%n",
                provider,
                e.getRetryAfterSeconds()
            );
        } catch (Exception e) {
            System.out.println("Error checking updates for " + provider + ": " + e.getMessage());
        }
    }

    /**
     * Resolves a provider name while keeping metadata failures local to that provider.
     *
     * @param client provider client whose name should be resolved.
     * @return provider name, or null when metadata lookup fails.
     */
    private String resolveProviderName(UpcomingLaunchClient client) {
        try {
            return client.getProviderName();
        } catch (Exception e) {
            System.out.println("Error checking updates for provider metadata: " + e.getMessage());
            return null;
        }
    }
}
